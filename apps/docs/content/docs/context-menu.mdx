## 安装

```bash
npx @joyui/cli add context-menu --yes
```

或通过 JoyUI MCP / CLI 安装。

## 图标与危险项

`icon`、`shortcut`；`variant="danger"` 用于破坏性操作。

<DemoFrame
  fileName="context-menu-basic.tsx"
  code={`import { RiDeleteBinLine, RiEditLine, RiShareLine } from "@remixicon/react";
import {
  ContextMenu, ContextMenuItem, ContextMenuSeparator, ContextMenuTrigger,
} from "@/components/base/context-menu/context-menu";

<ContextMenuTrigger>
  <div>右键打开</div>
  <ContextMenu aria-label="文件操作">
    <ContextMenuItem id="rename" icon={RiEditLine} shortcut="⌘R">重命名</ContextMenuItem>
    <ContextMenuItem id="share" icon={RiShareLine}>分享</ContextMenuItem>
    <ContextMenuSeparator />
    <ContextMenuItem id="delete" icon={RiDeleteBinLine} variant="danger">删除</ContextMenuItem>
  </ContextMenu>
</ContextMenuTrigger>`}
  canvasClassName="min-h-40 items-center justify-center bg-background-primary-default p-8"
>
  <ContextMenuBasicPreview />
</DemoFrame>

## 子菜单

`ContextMenuSubmenu` 嵌套二级选项。

<DemoFrame
  fileName="context-menu-submenu.tsx"
  code={`import { RiEditLine, RiFolderTransferLine } from "@remixicon/react";
import {
  ContextMenu, ContextMenuItem, ContextMenuSubmenu, ContextMenuTrigger,
} from "@/components/base/context-menu/context-menu";

<ContextMenuTrigger>
  <div>含子菜单</div>
  <ContextMenu aria-label="移动菜单">
    <ContextMenuItem id="rename" icon={RiEditLine}>重命名</ContextMenuItem>
    <ContextMenuSubmenu label="移动到" icon={RiFolderTransferLine}>
      <ContextMenuItem id="inbox">收件箱</ContextMenuItem>
      <ContextMenuItem id="archive">归档</ContextMenuItem>
    </ContextMenuSubmenu>
  </ContextMenu>
</ContextMenuTrigger>`}

## 应用场景

文件行右键：打开时描边高亮。

<DemoFrame
  fileName="context-menu-file.tsx"
  code={`import { RiDeleteBinLine, RiEditLine, RiShareLine } from "@remixicon/react";
import {
  ContextMenu, ContextMenuItem, ContextMenuSeparator, ContextMenuTrigger,
} from "@/components/base/context-menu/context-menu";

<ContextMenuTrigger>
  {({ isOpen }) => (
    <>
      <div>季度报告.pdf</div>
      <ContextMenu aria-label="文件操作">
        <ContextMenuItem id="rename" icon={RiEditLine}>重命名</ContextMenuItem>
        <ContextMenuItem id="delete" icon={RiDeleteBinLine} variant="danger">删除</ContextMenuItem>
      </ContextMenu>
    </>
  )}
</ContextMenuTrigger>`}
  canvasClassName="min-h-40 items-center justify-center bg-background-secondary-default p-8"
>
  <ContextMenuFileScenario />
</DemoFrame>
