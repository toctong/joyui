<ComponentDocPreview slug="prompt-suggestions" />

## 安装

```bash
npx @joyui/cli add prompt-suggestions --yes
```

## 选择回调

`onSelect` 收到点击的建议文案。

<DemoFrame
  fileName="prompt-suggestions-select.tsx"
  code={`import { PromptSuggestions } from "@/components/base/prompt-suggestions/prompt-suggestions";

<PromptSuggestions
  label="试试问"
  suggestions={["总结本周任务", "生成周报草稿"]}
  onSelect={(s) => console.log(s)}
/>`}
  canvasClassName="min-h-40 items-center justify-center bg-background-primary-default p-8"
>
  <PromptSuggestionsSelectPreview />
</DemoFrame>
