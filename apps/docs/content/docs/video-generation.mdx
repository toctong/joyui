## 安装

```bash
pnpm add @joyui/react
```

```tsx
import { VideoGeneration } from "@joyui/react";
// 或
import { VideoGeneration } from "@joyui/react/ai";
```

## 用法

```tsx
<VideoGeneration
  aspect="16:9"
  duration="10s"
  onAspectChange={setAspect}
  onDurationChange={setDuration}
  onSubmit={(prompt, meta) => renderClip(prompt, meta)}
  renderToolbar={({ aspectChips, durationChips }) => (
    <>
      {aspectChips}
      {durationChips}
      <YourExtraFilters />
    </>
  )}
/>
```
